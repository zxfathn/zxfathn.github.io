-- 1. Membuat Tabel tbkaryawan
CREATE TABLE `tbkaryawan` (
    `ktp` VARCHAR(20) NOT NULL PRIMARY KEY,
    `nama` VARCHAR(100) DEFAULT NULL,
    `jabatan` VARCHAR(100) DEFAULT NULL,
    `jenis_kelamin` VARCHAR(20) DEFAULT NULL,
    `agama` VARCHAR(20) DEFAULT NULL,
    `status` VARCHAR(30) DEFAULT NULL,
    `alamat` TEXT DEFAULT NULL,
    `password` VARCHAR(100) DEFAULT NULL
);

-- 2. Membuat Tabel tbpekerjaan
CREATE TABLE `tbpekerjaan` (
    `kodepekerjaan` VARCHAR(10) NOT NULL PRIMARY KEY,
    `namapekerjaan` VARCHAR(100) DEFAULT NULL,
    `jumlahtugas` INT(11) DEFAULT NULL
);

-- 3. Membuat Tabel tbgaji
CREATE TABLE `tbgaji` (
    `ktp` VARCHAR(20) DEFAULT NULL,
    `kode_pekerjaan` VARCHAR(10) DEFAULT NULL,
    `gaji_bersih` DOUBLE DEFAULT NULL,
    `gaji_kotor` DOUBLE DEFAULT NULL,
    `tunjangan` DOUBLE DEFAULT NULL,
    FOREIGN KEY (`ktp`) REFERENCES `tbkaryawan`(`ktp`),
    FOREIGN KEY (`kode_pekerjaan`) REFERENCES `tbpekerjaan`(`kodepekerjaan`)
);

-- 7. Query JOIN untuk mengambil data gabungan dari tbkaryawan, tbpekerjaan, dan tbgaji
SELECT 
    k.ktp, k.nama, k.jabatan, k.status, 
    p.namapekerjaan, g.gaji_bersih, g.gaji_kotor, g.tunjangan
FROM 
    tbkaryawan k
JOIN 
    tbgaji g ON k.ktp = g.ktp
JOIN 
    tbpekerjaan p ON g.kode_pekerjaan = p.kodepekerjaan;
