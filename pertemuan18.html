Fungsi dan Array di JavaScript

Contoh penggunaan Fungsi dan Array dalam JavaScript

Contoh Fungsi di JavaScript


function product(a, b) {
    return a * b;
}
document.write("Hasil perkalian 4 dan 3 adalah: " + product(4, 3));

Contoh Fungsi Array di JavaScript


var fruits = ["Apple", "Banana", "Mango"];
document.write("Fruits: " + fruits[0] + ", " + fruits[1] + ", " + fruits[2]);

Contoh Penggunaan Array dalam Konversi Suhu


function convertToCelsius(fahrenheit) {
    return (5/9) * (fahrenheit - 32);
}
var fahrenheitTemps = [212, 32, -459.15];
var celsiusTemps = [];
for (var i = 0; i < fahrenheitTemps.length; i++) {
    celsiusTemps[i] = convertToCelsius(fahrenheitTemps[i]);
}
document.write("Fahrenheit: " + fahrenheitTemps + "
Celsius: " + celsiusTemps);
